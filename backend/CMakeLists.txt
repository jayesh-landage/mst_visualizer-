cmake_minimum_required(VERSION 3.14)
project(mst_studio_backend LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(MST_BUILD_TESTS "Build GoogleTest unit tests" ON)
find_package(Threads REQUIRED)

# Header-only dependencies vendored in third_party/ (cpp-httplib, nlohmann/json)
add_library(mstcore STATIC
  src/DisjointSetUnion.cpp src/MSTResult.cpp src/Kruskal.cpp src/Prim.cpp src/Classifier.cpp
  src/GraphValidator.cpp src/Serialization.cpp src/ApiHandlers.cpp)
target_include_directories(mstcore PUBLIC include third_party)
target_link_libraries(mstcore PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(mstcore PUBLIC /W3 /bigobj)
  target_compile_definitions(mstcore PUBLIC _WIN32_WINNT=0x0A00)
else()
  target_compile_options(mstcore PUBLIC -Wall -Wextra)
endif()
if(WIN32)
  target_link_libraries(mstcore PUBLIC ws2_32)
endif()

add_executable(mst_server src/main.cpp)
target_link_libraries(mst_server PRIVATE mstcore)

if(MST_BUILD_TESTS)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/googletest EXCLUDE_FROM_ALL)
  enable_testing()
  add_executable(mst_tests tests/test_kruskal.cpp tests/test_prim.cpp tests/test_graph_validation.cpp tests/test_api.cpp)
  target_link_libraries(mst_tests PRIVATE mstcore gtest gtest_main)
  add_test(NAME mst_tests COMMAND mst_tests)
endif()
